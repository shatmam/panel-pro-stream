async function cargarTablaProveedores() { try { const res = await fetch("/api/proveedores", { headers: { "x-admin-key": localStorage.getItem("admin_key") } }); const data = await res.json(); const tbody = document.getElementById("body-proveedores"); // Cambia este ID por el de tu tbody tbody.innerHTML = ""; data.rows.forEach(r => { const tr = document.createElement("tr"); // Usamos el bucket del proveedor para el color (rojo/amarillo/verde) tr.className = `fila-${r.bucketProv}`; tr.innerHTML = ` ${r.servicio} ${r.correo} ${r.contrasena} ${r.diasProvTexto} ${r.venceProv} `; tbody.appendChild(tr); }); } catch (error) { console.error("Error cargando proveedores:", error); } } document.addEventListener("DOMContentLoaded", cargarTablaProveedores);
